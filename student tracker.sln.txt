
// ───────────── PIN DEFINITIONS ─────────────
#define enA 5
#define in1 6
#define in2 7
#define in3 8
#define in4 9
#define enB 10

#define ir_R A0
#define ir_F A1
#define ir_L A2

#define servo A4
#define pump A5

// ───────────── VARIABLES ─────────────
int Speed = 160;
int s1, s2, s3;

// ───────────── SETUP ─────────────
void setup() {
  Serial.begin(9600);

  pinMode(ir_R, INPUT);
  pinMode(ir_F, INPUT);
  pinMode(ir_L, INPUT);

  pinMode(enA, OUTPUT);
  pinMode(in1, OUTPUT);
  pinMode(in2, OUTPUT);
  pinMode(in3, OUTPUT);
  pinMode(in4, OUTPUT);
  pinMode(enB, OUTPUT);

  pinMode(servo, OUTPUT);
  pinMode(pump, OUTPUT);

  digitalWrite(pump, LOW);

  Serial.println("=== FIRE FIGHTING ROBOT START ===");

  // Servo calibration sweep
  Serial.println("Servo Calibration...");
  for (int angle = 90; angle <= 140; angle += 5) {
    Serial.print("Servo Angle: "); Serial.println(angle);
    servoPulse(servo, angle);
  }
  for (int angle = 140; angle >= 40; angle -= 5) {
    Serial.print("Servo Angle: "); Serial.println(angle);
    servoPulse(servo, angle);
  }
  for (int angle = 40; angle <= 90; angle += 5) {
    Serial.print("Servo Angle: "); Serial.println(angle);
    servoPulse(servo, angle);
  }

  analogWrite(enA, Speed);
  analogWrite(enB, Speed);

  Serial.println("System Ready.");
  delay(500);
}

// ───────────── LOOP ─────────────
void loop(){

  // Read sensors
  s1 = analogRead(ir_R);
  s2 = analogRead(ir_F);
  s3 = analogRead(ir_L);

  // Print sensor values
  Serial.print("R: "); Serial.print(s1);
  Serial.print(" | F: "); Serial.print(s2);
  Serial.print(" | L: "); Serial.print(s3);
  Serial.print(" || ");

  // ───────── DECISION LOGIC ─────────

  if(s1 < 250){
    Serial.println("🔥 ACTION: FIRE RIGHT");

    Stop();
    pumpON();

    Serial.println("Servo sweep RIGHT → CENTER");
    for(int angle = 90; angle >= 40; angle -= 3){
      servoPulse(servo, angle);
    }
    for(int angle = 40; angle <= 90; angle += 3){
      servoPulse(servo, angle);
    }
  }

  else if(s2 < 350){
    Serial.println("🔥 ACTION: FIRE FRONT");

    Stop();
    pumpON();

    Serial.println("Servo sweep FULL RANGE");
    for(int angle = 90; angle <= 140; angle += 3){
      servoPulse(servo, angle);
    }
    for(int angle = 140; angle >= 40; angle -= 3){
      servoPulse(servo, angle);
    }
    for(int angle = 40; angle <= 90; angle += 3){
      servoPulse(servo, angle);
    }
  }

  else if(s3 < 250){
    Serial.println("🔥 ACTION: FIRE LEFT");

    Stop();
    pumpON();

    Serial.println("Servo sweep LEFT → CENTER");
    for(int angle = 90; angle <= 140; angle += 3){
      servoPulse(servo, angle);
    }
    for(int angle = 140; angle >= 90; angle -= 3){
      servoPulse(servo, angle);
    }
  }

  else if(s1 >= 251 && s1 <= 700){
    Serial.println("➡️ ACTION: ADJUST RIGHT");

    pumpOFF();
    backword();
    delay(100);
    turnRight();
    delay(200);
  }

  else if(s2 >= 251 && s2 <= 800){
    Serial.println("⬆️ ACTION: MOVE FORWARD");

    pumpOFF();
    forword();
  }

  else if(s3 >= 251 && s3 <= 700){
    Serial.println("⬅️ ACTION: ADJUST LEFT");

    pumpOFF();
    backword();
    delay(100);
    turnLeft();
    delay(200);
  }

  else{
    Serial.println("⛔ ACTION: STOP (NO FIRE)");

    pumpOFF();
    Stop();
  }

  delay(50);
}

// ───────────── FUNCTIONS ─────────────

// Servo control
void servoPulse (int pin, int angle){
  int pwm = (angle * 11) + 500;
  digitalWrite(pin, HIGH);
  delayMicroseconds(pwm);
  digitalWrite(pin, LOW);
  delay(50);
}

// Pump control
void pumpON(){
  Serial.println("💧 Pump ON");
  digitalWrite(pump, HIGH);
}

void pumpOFF(){
  Serial.println("💧 Pump OFF");
  digitalWrite(pump, LOW);
}

// Motor controls
void forword(){
  Serial.println("🚗 Motor: FORWARD");
  digitalWrite(in1, HIGH);
  digitalWrite(in2, LOW);
  digitalWrite(in3, LOW);
  digitalWrite(in4, HIGH);
}

void backword(){
  Serial.println("🚗 Motor: BACKWARD");
  digitalWrite(in1, LOW);
  digitalWrite(in2, HIGH);
  digitalWrite(in3, HIGH);
  digitalWrite(in4, LOW);
}

void turnRight(){
  Serial.println("↪️ Motor: TURN RIGHT");
  digitalWrite(in1, LOW);
  digitalWrite(in2, HIGH);
  digitalWrite(in3, LOW);
  digitalWrite(in4, HIGH);
}

void turnLeft(){
  Serial.println("↩️ Motor: TURN LEFT");
  digitalWrite(in1, HIGH);
  digitalWrite(in2, LOW);
  digitalWrite(in3, HIGH);
  digitalWrite(in4, LOW);
}

void Stop(){
  Serial.println("🛑 Motor: STOP");
  digitalWrite(in1, LOW);
  digitalWrite(in2, LOW);
  digitalWrite(in3, LOW);
  digitalWrite(in4, LOW);
}
